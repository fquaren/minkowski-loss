\section{The setting in one page}
%=====================================================================

\paragraph{Fields and the two spaces.} A precipitation field is a vector $f\in\R_{\ge 0}^N$ of hourly accumulations
in mm, with $N = 128^2 = 16{,}384$ pixels of $2\times 2$\,km. Networks do not work in mm. They work in the
\emph{model space}
\begin{equation}
  z = \frac{\log(1+f)}{c},\qquad f = e^{cz}-1,\qquad c = \log(1+613\ \text{mm}) = 6.42,
\end{equation}
so $z\in[0,1]$. Two consequences of this choice come back later: $\log(1+f) = cz$ exactly, and
$\partial f/\partial z = c\,(1+f)$, which is large for heavy pixels.

\paragraph{The flow-matching residual model.} A deterministic U-Net (the \emph{backbone}) predicts a smooth field
$\mu$ in model space from the coarse input $X$. Flow matching (FM) then learns the \emph{residual}
\begin{equation}
  r = \frac{Y-\mu}{\sigma_r},\qquad \sigma_r = 0.01267,
\end{equation}
where $Y$ is the target in model space and $\sigma_r$ makes $r$ roughly unit-scale. FM learns a velocity field
$v_\theta(x,t,\mathbf c)$, with condition $\mathbf c = [X,\mu]$, that carries Gaussian noise to residuals. Training
draws $x_0\sim\mathcal N(0,I)$, $t\sim U(0,1)$, forms the straight path $x_t = (1-t)x_0 + t\,r$ and regresses the
velocity of that path:
\begin{equation}\label{eq:fm}
  \mathcal L_{\mathrm{FM}}(\theta) = \E\,\big\| v_\theta(x_t,t,\mathbf c) - (r - x_0)\big\|^2 .
\end{equation}
Sampling integrates $\mathrm dx/\mathrm dt = v_\theta$ from $t=0$ to $1$ with Heun's method, 16 steps of $h=1/16$:
\begin{equation}\label{eq:heun}
  x_{k+1} = x_k + \tfrac h2\Big[v_\theta(x_k,t_k) + v_\theta\big(x_k + h\,v_\theta(x_k,t_k),\,t_k+h\big)\Big].
\end{equation}
The generated field is $f = e^{c\,\mathrm{clamp}(\mu + \sigma_r x_{16},\,0,\,1)}-1$.

\section{The Minkowski functionals and the structural distance}
%=====================================================================

\paragraph{What is measured.} For a threshold $u$, the \emph{excursion set} is the set of pixels that reach it,
$E_u = \{i : f_i \ge u\}$. Three numbers describe its geometry: its area $A(u)$, its perimeter $P(u)$ and its
Euler characteristic $\chi(u)$ (number of pieces minus number of holes). In two dimensions these are, up to
linear combination, the only additive, continuous, motion-invariant functionals of a set
\citeq{Hadwiger's characterisation theorem}. The loss uses 15 thresholds from 0.1 to 79\,mm, evenly spaced in
$\xi = \log u$.

\paragraph{Discrete, differentiable estimators} (\texttt{src/losses/minkowski.py}). A hard indicator
$\mathbb 1[f_i\ge u]$ has zero derivative almost everywhere, so the loss uses a \emph{soft mask}
\begin{equation}\label{eq:mask}
  s_i(u) = \sigma\!\left(\frac{f_i-u}{\tau_u}\right),\qquad \tau_u = \max(0.1\cdot a\cdot u,\ 10^{-3}),
\end{equation}
with $\sigma$ the logistic function and anneal factor $a=0.05$, so $\tau_u = 0.005\,u$ (0.25\,mm at
$u = 50$\,mm). Then
\begin{align}
  A(u) &= \ell^2\sum_i \mathbb 1[f_i\ge u]\quad\text{(forward)},\qquad
         \frac{\partial A}{\partial f_i} := \ell^2\,\frac{\partial s_i}{\partial f_i}\quad\text{(backward)},\\
  P(u) &= \ell\Big[w_{\mathrm{ax}}\!\!\sum_{\text{axial pairs}}\!|s_i-s_j| + w_{\mathrm{di}}\!\!\sum_{\text{diagonal pairs}}\!|s_i-s_j|\Big],\\
  \chi(u) &= V - E_x - E_y + F ,
\end{align}
with $\ell = 2$\,km, Cauchy--Crofton weights $w_{\mathrm{ax}} = 0.473$, $w_{\mathrm{di}} = 0.218$, and $V,E,F$ the
soft counts of pixels, edges and faces (an edge or face counts with the minimum of its pixels' masks). The area
uses a \emph{straight-through estimator}: exact count forward, soft-mask derivative backward.

\paragraph{The curves and the distance.} Each field is summarised by three curves over the 15 thresholds,
\begin{equation}
  g(f) = \big[\log(1+A),\ \log(1+P),\ \operatorname{sign}(\chi)\log(1+|\chi|)\big] \in \R^{3\times 15},
\end{equation}
and two curves are compared by integrating the absolute difference over log-intensity,
\begin{equation}\label{eq:dist}
  d(g,h) = \frac13\sum_{k=1}^{3}\frac{1}{\xi_{15}-\xi_1}\int_{\xi_1}^{\xi_{15}} \big|g_k(\xi)-h_k(\xi)\big|\,\mathrm d\xi ,
\end{equation}
with the integral evaluated by the trapezoid rule on the thresholds. Every decade of intensity gets equal weight
in \emph{this sum}. Section~\ref{sec:heavy} shows that equal weight in the sum is not equal weight in the
gradient.

%=====================================================================
\section{How the loss is coupled to flow matching}\label{sec:coupling}
%=====================================================================

\paragraph{The problem the coupling solves.} The FM loss \eqref{eq:fm} lives on noisy intermediate states $x_t$,
which are not precipitation fields; the functionals only make sense on a finished field. So the structural loss
cannot simply be added to \eqref{eq:fm}. The fine-tune instead treats it as a \emph{reward} on generated samples,
following DRaFT-K \citep{clark2024draft}: generate a sample, score it, and back-propagate the score through the
sampler into $\theta$.

\paragraph{DRaFT-K with $K=1$.} Back-propagating through all 16 Heun steps would store 32 network evaluations and
let the credit for the final sample spread thinly over the whole path. DRaFT-K cuts the graph: the first $16-K$
steps run with gradients switched off. With $K = 1$,
\begin{equation}
  x_{15} = \sg\big(\text{15 Heun steps from } x_0\big),\qquad x_{16} = \Phi_\theta(x_{15}),\qquad
  \nabla_\theta R = \frac{\partial R}{\partial x_{16}}\;\frac{\partial \Phi_\theta(x_{15})}{\partial\theta},
\end{equation}
where $\sg$ stops the gradient and $\Phi_\theta$ is the last Heun step \eqref{eq:heun}. Only the last
$1/16$ of the path is trained by the reward. \status{inference} In FM the last step mostly sets fine detail and
amplitude, so a $K=1$ reward acts on the smallest scales, which is where both Minkowski-coupled models later
showed excess spectral power. (The coupling check found the $K=1$ and $K=2$ gradients aligned, cosine
$+0.97$, \status{measured}, \texttt{logs/coupling\_v4adv\_*.log}.)

\paragraph{One optimiser step.} For a batch of $B = 8$ conditions, draw $M$ members per condition, decode each to
mm, compute $g$ for each member, and compare with the target curve $g^{\mathrm{obs}}$ (precomputed from the
observed field). The total loss is
\begin{equation}\label{eq:total}
  L = w_{\mathrm r}\,R + w_{\mathrm p}\,\mathcal L_{\mathrm{prox}} + w_{\mathrm{ret}}\,\mathcal L_{\mathrm{FM}},
\end{equation}
optimised with AdamW, learning rate $10^{-5}$, gradient clipping at norm 1, 500 steps per epoch for 10 epochs.

\subsection{The distributional reward}
\begin{equation}\label{eq:dist-reward}
  R_{\mathrm{dist}} = \frac1B\sum_{b} d\Big(\tfrac1M\textstyle\sum_m g_{bm},\ g^{\mathrm{obs}}_b\Big)
  + \lambda_s\, d\Big(\log\!\big(1+\operatorname{std}_{b,m}\, g_{bm}\big),\ \log\!\big(1+\operatorname{std}_b\, g^{\mathrm{obs}}_b\big)\Big).
\end{equation}
The first term asks the \emph{average} member curve of each condition to match the observation. The second asks
the spread of the curves to match the spread of the observed curves.

\paragraph{What the spread term actually measures.} \status{derivation} Both standard deviations are pooled over
the batch: $\operatorname{std}_{b,m}$ over all $M\times B = 32$ generated curves. By the law of total variance,
\begin{equation}
  \operatorname{Var}_{b,m}(g) = \underbrace{\E_b\big[\operatorname{Var}_m(g\mid b)\big]}_{\text{ensemble spread}}
  + \underbrace{\operatorname{Var}_b\big(\E_m[g\mid b]\big)}_{\text{differences between patches}} .
\end{equation}
Eight different patches differ far more from each other than four members of one patch do, so the second term
dominates. As written, the spread term mostly asks the eight patches to differ as much as their targets do, and
does little to control the ensemble's own spread.

\subsection{The energy reward}
\begin{equation}\label{eq:energy}
  R_{\mathrm{ES}} = \frac1B\sum_b\Big[\frac1M\sum_m d\big(g_{bm}, g^{\mathrm{obs}}_b\big)
   - \frac{1}{2M(M-1)}\sum_{m\ne m'} d\big(g_{bm}, g_{bm'}\big)\Big].
\end{equation}
The first term pulls every member toward the observation; the second rewards members that differ from each
other. This is an unbiased estimate of the \emph{energy score}
$\mathrm{ES}(P, y) = \E_P\, d(G, y) - \tfrac12\E_P\, d(G, G')$, with $G, G'$ independent draws from the model
\citeq{Gneiting \& Raftery 2007, energy score; publisher page not reachable this session}.

\paragraph{Why $M(M-1)$ and not $M^2$.} \status{derivation} For independent members,
\[
  \E\Big[\frac{1}{M(M-1)}\sum_{m\ne m'} d(G_m,G_{m'})\Big] = \E\, d(G,G').
\]
Dividing the same sum by $M^2$ gives
$\frac{M-1}{M}\E\,d(G,G')$ instead: the bonus for spread shrinks, and the minimiser becomes an ensemble that is too
narrow, which is the failure the reward is meant to correct.

\paragraph{Why the energy score is the principled choice.} A scoring rule is \emph{proper} if its expected value
is smallest when the model's distribution equals the true one. For a distance $d$ of negative type, the energy
score is proper: the gap between using $P$ and the truth $Q$ is half the energy distance between them, which is
non-negative. The trapezoid integral \eqref{eq:dist} is a weighted $L^1$ distance between 45-dimensional vectors,
and $L^1$ distances are of negative type, so $R_{\mathrm{ES}}$ is proper for the distribution of the curves
$g$ \status{derivation, given the negative-type property}. Whether it is \emph{strictly} proper (the truth is the
unique minimiser) with an $L^1$ distance, we do not know \citeq{strong negative type of $\ell^1$ metrics}. And
proper only ever means proper \emph{on the features}: two field distributions with the same distribution of
curves score the same.

\subsection{The proximal anchor and the retention term}
\begin{equation}
  \mathcal L_{\mathrm{prox}} = \big\|v_\theta(\tilde x_t, t,\mathbf c) - v_{\mathrm{ref}}(\tilde x_t,t,\mathbf c)\big\|^2,
  \qquad \tilde x_t = (1-t)\,x_0 + t\,\tilde x_1,
\end{equation}
where $v_{\mathrm{ref}}$ is the frozen clean model. It should keep the fine-tuned velocity close to the clean one
where it matters. The retention term is the ordinary FM loss \eqref{eq:fm} on the true residual; it supplies a
pixel-by-pixel data term, which the reward cannot, because the functionals do not change when a field is shifted
or rotated.

\section{Three problems found in the coupling}\label{sec:problems}
%=====================================================================

\subsection{The energy fine-tune is mostly more flow-matching training}
By value, the terms of \eqref{eq:total} in the energy run are $w_{\mathrm r}R\approx 10^{-3}\times 0.20 =
2\times10^{-4}$ and $w_{\mathrm{ret}}\mathcal L_{\mathrm{FM}}\approx 0.24$. Values are not gradients, so we
measured the gradient of each term separately on four training batches at the clean checkpoint
(\texttt{scripts/evaluate/grad\_norm\_terms.py}, \texttt{logs/grad\_norm\_terms\_v4adv.txt}) \status{measured}:

run & reward & anchor & FM retention & reward share & $\cos(\nabla R,\nabla L)$\\\midrule
energy & $8.3\times10^{-4}$ & $1.2\times10^{-4}$ & $0.65$ & 0.13\% & $-0.01$\\
dist & $8.3\times10^{-4}$ & $0.7\times10^{-4}$ & 0 & 99\% & $+0.99$\\\bottomrule
\end{tabular}\\[2pt]
{\footnotesize Norms of $\nabla_\theta$ of $w_{\mathrm r}R$, $w_{\mathrm p}\mathcal L_{\mathrm{prox}}$ and
$w_{\mathrm{ret}}\mathcal L_{\mathrm{FM}}$, before clipping; share $=\|\nabla(w_{\mathrm r}R)\|/\|\nabla L\|$.}
\end{center}

\paragraph{Why the absolute size does not matter but the mix does.} \status{derivation} Adam's update is
$\Delta\theta = -\eta\,\hat m/(\sqrt{\hat v}+\epsilon)$, with $\hat m,\hat v$ running averages of the gradient and
its square. Multiplying every gradient by a constant $\lambda$ multiplies $\hat m$ by $\lambda$ and $\sqrt{\hat v}$ by
$|\lambda|$, so the step is unchanged (as long as $\epsilon$ is negligible). The dist run's tiny total norm
($8\times10^{-4}$, far below the clip at 1) therefore still produces full-size steps, all in the reward's direction.
In the energy run the step direction is the FM loss's: the reward is 0.13\% of the gradient and orthogonal to the
total. The energy fine-tune is, to a good approximation, ten more epochs of FM training at a low learning rate
(its \texttt{train\_fm} fell from 0.2431 to 0.2395). Any difference between FM + energy and clean FM cannot be
credited to the reward without the control run with $w_{\mathrm r} = 0$.

\subsection{The anchor is evaluated where the sampler never goes}
The code sets $\tilde x_1 = \mathrm{clamp}(f/6.42,\,0,\,1)$: the generated field \emph{in mm}, divided by $c$ and
saturating at 6.4\,mm. But the model's coordinate is the scaled log-residual $r = (Y-\mu)/\sigma_r$, mostly near 0
and of order $\pm1$, with outliers to $|r|\approx 34$. \begin{example}
A pixel with 10\,mm gives $\tilde x_1 = \min(10/6.42, 1) = 1$ wherever it is. A residual of $0.05$ in model space
(the backbone 0.05 below the target, i.e.\ a factor $e^{0.05c}\approx1.38$ in $1+f$) gives $r = 0.05/0.01267 \approx 3.9$.
\end{example}
\noindent So $\tilde x_t$ blends noise with a binary-looking rain mask, states the ODE does not visit, and the anchor
constrains the velocity there, not along the model's real paths \status{derivation from the code,
\texttt{reward\_finetune.py:348--353}}. In the dist run, which has no retention term, this anchor is the only thing
holding the model near the clean one. \status{inference} That fits the dist model's drift toward small-scale
speckle in light rain, which the reward does not penalise.

\subsection{The dist reward barely moved during training}
Training reward 0.351 $\to$ 0.339 over 10 epochs; validation reward 0.361 $\to$ 0.350 with swings of $\pm0.015$
between epochs. The coupling check put the per-batch noise at 11\% of the mean, so 20 validation batches carry
a noise of about $\pm0.008$ on the mean; the total drop is about 1.4 times that \status{measured}. The model did
change (the evaluations show it), but the objective it was trained on improved by an amount near its noise. Either
the change is many small steps whose average helps the tail while each batch is too noisy to show it, or it is drift
the anchor did not catch. This is why the gain of the distributional fine-tune is not yet attributed.

\section{Why the reward misses heavy rain}\label{sec:heavy}
%=====================================================================

\paragraph{One pixel, one threshold.} \status{derivation} Through the soft mask \eqref{eq:mask}, pixel $i$ feels
threshold $u$ only through
\begin{equation}\label{eq:sens}
  \frac{\partial s_i}{\partial f_i} = \frac{1}{\tau_u}\,\sigma'\!\left(\frac{f_i-u}{\tau_u}\right),\qquad
  \sigma'(a) = \sigma(a)\big(1-\sigma(a)\big)\approx e^{-|a|}\ \text{for}\ |a|\gg 1 .
\end{equation}
The kernel has width $\tau_u = 0.005u$: a pixel feels only the thresholds within a few per cent of its own value.

\begin{example}
A predicted core of 17\,mm under a target that exceeds 20.2, 31.7 and 50.1\,mm. For $u = 20.2$\,mm,
$\tau = 0.101$\,mm and $a = (17-20.2)/0.101 = -31.7$, so $\partial s/\partial f \approx e^{-31.7}/0.101 \approx
1.7\times10^{-13}$\,mm$^{-1}$. For $u = 50.1$, $\tau = 0.25$\,mm, $a = -132$ and $e^{-132}\approx 5\times10^{-58}$, which underflows to zero in single precision. The thresholds where
the error is largest are the ones that give no gradient.
\end{example}

\paragraph{All pixels, one threshold: the pixel-mass limit.} \status{derivation} Summing \eqref{eq:sens} over
pixels and writing $p$ for the density of pixel values,
\begin{equation}
  \sum_i \frac{\partial s_i}{\partial f_i} = N\!\int p(f)\,\frac{1}{\tau}\sigma'\!\Big(\frac{f-u}{\tau}\Big)\mathrm df
  \;\longrightarrow\; N\,p(u)\quad (\tau\to0),
\end{equation}
because $\tau^{-1}\sigma'(\cdot/\tau)$ is a probability density of width $\sim\tau$ (it integrates to 1). A
threshold's total pull is the number of pixels per mm at $u$, \emph{whatever $\tau$ is}. Two results follow:
\begin{itemize}[nosep]
\item Light rain has most pixels, so it takes most of the gradient: on the clean hourly FM, 73\% of $|\partial R/\partial z|$
lands on pixels below 1\,mm on the heavy subset and 89\% on the random subset (Fig.~\ref{fig:share}, blue)
\status{measured}. On 15-min data the same limit was measured as ``83\% of the gradient from thresholds
$\le 0.28$\,mm/h'' (DECISIONS \S5).
\item Changing the anneal factor changes $\tau$, so it cannot change the shares to leading order. This explains
the earlier sweep (DECISIONS \S5): anneal 0.05 $\to$ 0.40 raised the absolute tail gradient by eleven orders of
magnitude while the tail's share stayed at 0.75--0.88\%.
\end{itemize}

\paragraph{The misconception to drop.} ``Equal weight per decade in the loss means equal weight per decade in
training.'' The distance \eqref{eq:dist} weights decades equally, but the gradient of each decade's term is
multiplied by the number of pixels near it. Equal weight in the objective is not equal pull on the model.
The intuition is right for a quantity that is linear in the curve values; it fails here because the curves'
sensitivity to the field is itself proportional to pixel mass.

\section{Options and the decision}
%=====================================================================

The two failures need different fixes: (1) the gradient grows with pixel mass; (2) a threshold out of the
prediction's reach gives nothing. All options keep $K = 1$.

\begin{description}[leftmargin=1.2em]
\item[A. Inverse-mass weights.] Weight each threshold's term by $(n_j+1)^{-\alpha}$, where $n_j$ is the target's
number of pixels above $u_j$. Fixes (1), not (2). A few lines.
\item[B. Hard forward, heavy-tailed backward.] Keep exact counts forward and replace $\sigma'$ in the backward pass
by a heavy-tailed kernel (e.g.\ Cauchy) of width $\sim0.3u$ at high thresholds. Fixes (2) with A for (1). Needs the
same straight-through treatment for $P$ and $\chi$, adds a width hyperparameter, and a surrogate gradient can point
where the true objective does not, which is how gaming starts.
\item[C. Quantile form of the area.] Compare the area curve horizontally instead of vertically (next section).
Removes (2) exactly for the area channel, with no new hyperparameter. Needs A for $P$ and $\chi$.
\end{description}

\noindent\textbf{Decision (researcher, 2026-10-29): C for the area, A for perimeter and Euler.} In the code this
is \texttt{REWARD\_FORM: quantile\_mass}; the original stays available as \texttt{REWARD\_FORM: threshold}, the
default.

\section{The quantile area}\label{sec:quantile}
%=====================================================================

\paragraph{Same curve, read along the other axis.} The area curve is a \emph{survival function}:
$S_f(v) = \#\{i: f_i\ge v\}$, so $A(u) = \ell^2 S_f(u)$. Plot it with intensity on the horizontal axis and pixel
count on the vertical axis (Fig.~\ref{fig:survival}). The original loss compares the two curves
\emph{vertically}: at a fixed intensity $u_j$, how many pixels each field has above it. One can equally compare
them \emph{horizontally}: at a fixed count $n$, which intensity each field reaches. The intensity at which a field
has exactly $n$ pixels above it is its $n$-th largest value, written $f_{(n)}$.

\begin{figure}[t]\centering
\includegraphics[width=0.9\linewidth]{figures/reward_coupling/survival.pdf}
\caption{Survival curves of a real heavy validation patch (patch 979, target maximum 87\,mm) and one clean-FM member.
Vertically, at $u = 50.1$\,mm, the target has 7 pixels above the threshold and the prediction 1; the vertical term
compares $\log(1+A)$ values and its gradient needs a predicted pixel near 50.1\,mm. Horizontally, at the same count
$n = 7$, the target reaches 53.7\,mm and the prediction 31.2\,mm; this difference has a gradient on the predicted
pixel of rank 7 wherever its value is.}\label{fig:survival}
\end{figure}

\paragraph{The new area channel.} For condition $b$ and threshold $u_j$, let $n_j = \#\{i : y_i\ge u_j\}$ be the
target's count. The area entry of the curve becomes
\begin{equation}
  q_j(f) = \log\big(1 + f_{(n_j)}\big),\qquad q_j(y) = \log\big(1 + y_{(n_j)}\big),
\end{equation}
masked (weight 0) where $n_j = 0$, i.e.\ above the target maximum. Since $y_{(n_j)}$ is the smallest target value
that is still $\ge u_j$, the target side sits just above each threshold.

\paragraph{Its gradient.} \status{derivation} In model space $\log(1+f) = cz$, so $q_j(f) = c\,z_{(n_j)}$ and
\begin{equation}
  \frac{\partial}{\partial z_i}\big|q_j(f) - q_j(y)\big| = \pm c\quad\text{for the pixel of rank } n_j,\quad 0\ \text{otherwise.}
\end{equation}
Every populated threshold moves exactly one pixel per member, with the same magnitude $c$, whatever that pixel's
value. (Sorting is differentiable in the sense needed: the sorted values are a permutation of the inputs, and the
gradient flows to the selected element.) In Fig.~\ref{fig:survival} the term at $n = 7$ pushes the predicted
31.2\,mm pixel up toward 53.7\,mm, where the vertical term gives nothing.

\paragraph{Link to Wasserstein distances, and where it breaks.} For two sets of $N$ values, the 1-D Wasserstein
distance is $W_1 = \frac1N\sum_n |f_{(n)} - y_{(n)}|$: a horizontal comparison of the two distributions at every
rank. Arjovsky et al.\ make the corresponding argument for distributions: a horizontal (transport) distance gives a
usable gradient where a vertical comparison saturates because the supports do not overlap \citep{arjovsky2017wgan}.
Our area term is a horizontal comparison at 15 ranks only, in log-intensity, per patch. The analogy breaks in two
places: only $\le15$ pixels per member get a gradient (not all $N$), and the comparison is per patch, not between
distributions of patches.

\section{Inverse-mass weights for perimeter and Euler}
%=====================================================================

Perimeter and Euler stay vertical (they have no simple horizontal form) and are weighted per threshold:
\begin{equation}
  w_j = \frac{(n_j+1)^{-\alpha}}{\frac1{15}\sum_{j'}(n_{j'}+1)^{-\alpha}},\qquad \alpha = 1,
\end{equation}
so the weights average 1. Thresholds above the target maximum ($n_j = 0$) take the weight of the highest populated
threshold, so that overshoot is penalised as strongly as the top of the target, not infinitely. The weighted
distance replaces $|g_k-h_k|$ by $w\,|g_k-h_k|$ inside \eqref{eq:dist}.

\paragraph{Why $\alpha = 1$.} \status{derivation under an assumption} The pull of threshold $u$ is the number of
pixels near it. Suppose the survival curve is locally a power law, $S(v)\propto v^{-k}$. The number of pixels per
unit of log-intensity at $u$ is then $-\mathrm dS/\mathrm d\log v = k\,S(u)\propto n_j$. Weighting by $1/n_j$
cancels it. The assumption fails where the survival curve bends (the flat top below 1\,mm in
Fig.~\ref{fig:survival}), and the perimeter and Euler sensitivities depend on boundary pixels rather than on all
pixels near $u$, so we checked $\alpha\in\{0, 0.5, 1\}$ directly (below) and $\alpha = 1$ gave the most even
spread across intensity bins.

\paragraph{Properness is kept.} \status{derivation} The ranks $n_j$ and the weights $w_j$ depend only on the
target. For each condition the reward therefore compares fixed feature vectors with a fixed weighted $L^1$
distance (non-negative weights keep the negative-type property; zero weights make it a pseudometric on the
features). The energy score stays proper for the distribution of these features. If the weights depended on the
prediction, this would fail.

\section{Checks before any fine-tune}
%=====================================================================

\paragraph{Unit tests} (\texttt{tests/test\_reward\_form.py}, 13 tests, all pass; on the previous code the file
fails at import) \status{measured}:
\begin{itemize}[nosep]
\item a 17\,mm predicted core under an 80\,mm target: the original form gives the core no gradient, the new form
pulls it up (negative $\partial R/\partial f$), in all three modes;
\item unit weights reproduce the original reward exactly, in all three modes;
\item the area feature equals $\log(1+f_{(n_j)})$ and is masked where $n_j = 0$; an exact prediction scores 0;

\paragraph{Gradient split per term for the new form} \status{measured}: with the new form the reward's gradient norm is
$5.0\times10^{-4}$ in both modes (was $8.3\times10^{-4}$). In the dist configuration it is 96\% of the total
(cos $+0.97$). In the energy configuration it is \textbf{0.07\%} of the total (cos $-0.03$): the retention term still
sets the direction of every step. A fine-tune of the energy configuration with the new form would therefore again be
continued FM training. Before running one, the reward and retention weights must be rebalanced, e.g.\ by matching
their gradient norms (the gradient-norm rule of DECISIONS \S9), which here means raising $w_{\mathrm r}$ by roughly
three orders of magnitude or lowering $w_{\mathrm{ret}}$ accordingly.

%=====================================================================
\section{Costs, risks and what is not done}
%=====================================================================

\begin{itemize}[leftmargin=1.2em]
\item \textbf{Noisier reward.} Giving rare thresholds weight raises the reward's sampling noise (distributional:
11.4\% $\to$ 18.9\% of the mean). Training-time changes will be harder to see on 20 validation batches.
\item \textbf{Artefacts get more weight.} The top populated threshold may hold a handful of pixels, and the top of
the distribution contains radar artefacts. The new reward pulls harder toward them. Look at images of anything it
produces.
\item \textbf{Sparse area gradient.} At most 15 pixels per member per step from the area term. The overall field
gradient was already concentrated before (100\% of its mass on the top 1\% of pixels in both forms).
\item \textbf{Properness only on the features.} The reward cannot see anything the 45 numbers per field miss,
including position: the functionals carry no positional information, so placement must come from a data term.
\item \textbf{Not a result yet.} No fine-tune has run with the new form. Before one, two items in EXPERIMENTS \S5:
fix the proximal anchor so it is evaluated on the model's own coordinates, and run the $w_{\mathrm r}=0$ control to
attribute the energy run. For the energy configuration, also rebalance $w_{\mathrm r}$ against $w_{\mathrm{ret}}$:
as configured, its reward is 0.07\% of the gradient with either form.
\item \textbf{Where the picture stops holding.} The pixel-mass argument assumes $p$ varies slowly over $\tau$; the
$\alpha=1$ argument assumes a locally power-law survival curve; and all gradient measurements are at the clean
checkpoint, before any fine-tuning has moved the model.
\end{itemize}
